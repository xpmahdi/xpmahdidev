# Python Data Ingestion Worker
import httpx
import psycopg2
from psycopg2.extras import execute_batch

def ingest_media_records(api_endpoint: str, db_conn):
    response = httpx.get(api_endpoint, timeout=10.0)
    records = response.json().get("items", [])
    
    clean_payload = [
        (item["id"], item["title"], item.get("release_year"), item.get("rating"))
        for item in records if item.get("title")
    ]
    
    query = """
        INSERT INTO media_catalog (external_id, title, release_year, rating)
        VALUES (%s, %s, %s, %s)
        ON CONFLICT (external_id) DO UPDATE 
        SET rating = EXCLUDED.rating, updated_at = NOW();
    """
    with db_conn.cursor() as cur:
        execute_batch(cur, query, clean_payload, page_size=200)
    db_conn.commit()
    return len(clean_payload)
